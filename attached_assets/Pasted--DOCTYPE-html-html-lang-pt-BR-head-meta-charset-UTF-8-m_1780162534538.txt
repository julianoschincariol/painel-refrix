// ====== HELPER: format date ======
function now() {
    return new Date().toLocaleString('pt-BR', { timeZone: 'America/Sao_Paulo' });
}
function formatTimestamp() {
    return new Date().toLocaleString('pt-BR', { timeZone: 'America/Sao_Paulo', day:'2-digit', month:'2-digit', year:'numeric', hour:'2-digit', minute:'2-digit', second:'2-digit' });
}
function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

// ====== PROXY FETCH ======
async function fetchWithProxy(url, useProxy = true, timeout = 12000) {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), timeout);
    try {
        let finalUrl = url;
        if (useProxy) {
            finalUrl = `https://api.allorigins.win/raw?url=${encodeURIComponent(url)}`;
        }
        const resp = await fetch(finalUrl, { signal: controller.signal });
        clearTimeout(timer);
        if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
        return await resp.text();
    } catch (e) {
        clearTimeout(timer);
        throw e;
    }
}

// ====== API FETCH FUNCTIONS ======
async function fetchIPCA() {
    // BCB série 13522 (IPCA 12m) - último valor
    const data = await fetchWithProxy('https://api.bcb.gov.br/dados/serie/bcdata.sgs.13522/dados/ultimos/1?formato=json', false);
    const json = JSON.parse(data);
    if (!json || !json.length) throw new Error('Empty');
    const val = parseFloat(json[0].valor.replace(',','.'));
    return { value: val, raw: val, unit: '%', label: val.toFixed(2).replace('.',',') + '%' };
}

async function fetchSelic() {
    // BCB série 432 (Selic meta)
    const data = await fetchWithProxy('https://api.bcb.gov.br/dados/serie/bcdata.sgs.432/dados/ultimos/1?formato=json', false);
    const json = JSON.parse(data);
    if (!json || !json.length) throw new Error('Empty');
    const val = parseFloat(json[0].valor.replace(',','.'));
    return { value: val, raw: val, unit: '%', label: val.toFixed(2).replace('.',',') + '%' };
}

async function fetchDolar() {
    // BCB OData PTAX
    const hoje = new Date();
    const mm = String(hoje.getMonth()+1).padStart(2,'0');
    const dd = String(hoje.getDate()).padStart(2,'0');
    const yyyy = hoje.getFullYear();
    const dataStr = `${mm}-${dd}-${yyyy}`;
    const url = `https://olinda.bcb.gov.br/olinda/servico/PTAX/versao/v1/odata/CotacaoDolarDia(dataCotacao=@dataCotacao)?@dataCotacao='${dataStr}'&$top=1&$format=json`;
    const text = await fetchWithProxy(url, true, 10000);
    const json = JSON.parse(text);
    if (!json || !json.value || !json.value.length) throw new Error('Sem cotação para hoje');
    const cotacao = json.value[0].cotacaoVenda;
    const val = parseFloat(cotacao);
    return { value: val, raw: val, unit: 'R$', label: `R$ ${val.toFixed(2).replace('.',',')}` };
}

async function fetchBrent() {
    // brapi.dev BZ=F
    const text = await fetchWithProxy('https://brapi.dev/api/quote/BZ=F?token=none', true, 10000);
    const json = JSON.parse(text);
    if (!json || !json.results || !json.results.length) throw new Error('No results');
    const val = json.results[0].regularMarketPrice;
    if (val === undefined || val === null) throw new Error('No price');
    return { value: val, raw: val, unit: 'US$/bbl', label: `US$${val.toFixed(2).replace('.',',')}` };
}

async function fetchDiesel() {
    // ANP Tenta buscar CSV mensal (exemplo genérico)
    const url = 'https://www.gov.br/anp/pt-br/centrais-de-conteudo/dados-abertos/arquivos/precos-semestrais/precos-semestrais.csv';
    const text = await fetchWithProxy(url, true, 12000);
    // Parse rápido: procurar linha com "DIESEL S10"
    const lines = text.split('\n');
    let val = null;
    for (const line of lines) {
        if (line.toLowerCase().includes('diesel s10') || line.toUpperCase().includes('DIESEL S10')) {
            // espera formato: ...;7,24;...
            const parts = line.split(';');
            for (let p of parts) {
                p = p.trim().replace(',','.');
                if (!isNaN(parseFloat(p)) && parseFloat(p) > 1) { val = parseFloat(p); break; }
            }
            if (val) break;
        }
    }
    if (!val) throw new Error('Preço não encontrado no CSV');
    return { value: val, raw: val, unit: 'R$/L', label: `R$ ${val.toFixed(2).replace('.',',')}` };
}

async function fetchAcucar() {
    // CEPEA - widget JS (tenta JSONP - simulação com proxy)
    const url = 'https://www.cepea.esalq.usp.br/br/indicador/acucar-cristal.aspx?output=json';
    const text = await fetchWithProxy(url, true, 12000);
    // Tenta parse de um objeto js (pode ser JSONP)
    let json;
    try {
        const cleaned = text.replace(/^.*?\(/,'').replace(/\)\s*$/,'').trim();
        json = JSON.parse(cleaned);
    } catch(e) {
        // Tenta parse direto
        json = JSON.parse(text);
    }
    if (json && json.preco) {
        const val = parseFloat(json.preco.replace(',','.'));
        return { value: val, raw: val, unit: 'R$/sc', label: `R$ ${val.toFixed(2).replace('.',',')}` };
    }
    throw new Error('Formato inesperado');
}

    { id:'diesel', name:'Diesel S10', fallback: FALLBACKS.diesel, fetchFn: fetchDiesel, manual: false },
    { id:'acucar', name:'Açúcar Cristal SP', fallback: FALLBACKS.acucar, fetchFn: fetchAcucar, manual: false },
    { id:'pet', name:'PET Virgem', fallback: FALLBACKS.pet, fetchFn: null, manual: true },
    { id:'icc', name:'ICC FGV', fallback: FALLBACKS.icc, fetchFn: null, manual: true }
];

indicatorDefs.forEach(def => {
    indicators[def.id] = {
        ...def,
        current: def.fallback,
        status: 'yellow', // green = api ok, yellow = fallback, red = failed
        lastUpdate: null,
        apiStatus: def.manual ? '⚙️ Manual' : '⏳ Inicializando...'
    };
    // For manual ones, set green
    if (def.manual) {
        indicators[def.id].status = 'green';
        indicators[def.id].apiStatus = '✅ Manual';
        indicators[def.id].lastUpdate = now();
    }
});

// ====== UPDATE UI ======
function renderCards() {
    const grid = document.getElementById('cardsGrid');
    grid.innerHTML = '';
    const ids = ['ipca','selic','brent','dolar','diesel','acucar','pet','icc'];
    for (const id of ids) {
        const ind = indicators[id];
        const val = ind.current;
        const card = document.createElement('div');
        card.className = `card ${ind.status}`;
        card.id = `card-${id}`;
        let statusText = '';
        if (ind.apiStatus.includes('✅')) statusText = '✅ OK';
        else if (ind.apiStatus.includes('⚠️') || ind.apiStatus.includes('Parcial')) statusText = '⚠️ Parcial';
        else if (ind.apiStatus.includes('❌')) statusText = '❌ Erro';
        else if (ind.manual) statusText = '⚙️ Manual';
        else statusText = ind.apiStatus;
        const timeStr = ind.lastUpdate ? ind.lastUpdate : '—';
        card.innerHTML = `
            <div class="card-header">
                <span class="card-title">${ind.name}</span>
                <span class="card-status">${statusText}</span>
            </div>
            <div class="card-value">${val.label}</div>
            <div class="card-unit">${val.unit}</div>
            <div class="card-time">
                <span>🕒 ${timeStr}</span>
                <span>${ind.manual ? '✏️ Manual' : '📡 Automático'}</span>
            </div>
        `;
        grid.appendChild(card);
    }
}

function updateLeituraDetalhada() {
    const nApis = indicatorDefs.filter(d => !d.manual).length;
    let ok = 0, parcial = 0, erro = 0;
    for (const id in indicators) {
        const ind = indicators[id];
        if (ind.manual) continue;
        if (ind.apiStatus.includes('✅')) ok++;
        else if (ind.apiStatus.includes('⚠️') || ind.apiStatus.includes('Parcial')) parcial++;
        else if (ind.apiStatus.includes('❌')) erro++;
    }
    const statusText = `${ok}/${nApis} OK, ${parcial} parcial, ${erro} erro`;
    document.getElementById('statusGeral').textContent = statusText;
    document.getElementById('ultimaAtualizacao').textContent = lastFullRefresh ? lastFullRefresh : '—';
    document.getElementById('modoInfo').textContent = offlineMode ? 'Offline (fallback)' : 'Online (real)';
}

function updateTicker() {
    const ticker = document.getElementById('ticker');
    const ipca = indicators.ipca.current.label;
    const selic = indicators.selic.current.label;
    const dolar = indicators.dolar.current.label;
    ticker.textContent = `📊 IPCA ${ipca} | Selic ${selic} | Dólar ${dolar} ${now()}`;
}

function showToast(msg, duration = 3000) {
    const t = document.getElementById('toast');
    t.textContent = msg;
    t.style.display = 'block';
    setTimeout(() => t.style.display = 'none', duration);
}

// ====== FETCH ALL ======
async function fetchAllIndicators() {
    const promises = [];
    for (const def of indicatorDefs) {
        if (def.fetchFn) {
            promises.push(
                def.fetchFn()
                    .then(result => {
                        indicators[def.id].current = result;
                        indicators[def.id].status = 'green';
                        indicators[def.id].apiStatus = '✅ OK';
                        indicators[def.id].lastUpdate = now();
                        console.log(`✅ ${def.name}: ${result.label}`);
                    })
                    .catch(err => {
                        console.warn(`❌ ${def.name}: ${err.message || err}`);
                        // Usa fallback
                        const fallback = def.fallback;
                        indicators[def.id].current = fallback;
                        indicators[def.id].status = 'yellow';
                        indicators[def.id].apiStatus = '⚠️ Parcial (fallback)';
                        indicators[def.id].lastUpdate = now();
                        // Se já tinha fallback e não veio da API, mantém amarelo
                        if (offlineMode) {
                            indicators[def.id].apiStatus = '⚠️ Offline (fallback)';
                        } else {
                            indicators[def.id].apiStatus = '⚠️ Fallback';
                        }
                    })
            );
        }
    }
    await Promise.allSettled(promises);

    // Verifica se todas as APIs falharam -> modo offline
    const apisNaoManual = indicatorDefs.filter(d => !d.manual);
    const anySucesso = apisNaoManual.some(d => indicators[d.id].apiStatus.includes('✅'));
    if (!anySucesso) {
        offlineMode = true;
        showToast('⚠️ Modo offline ativado – APIs indisponíveis', 5000);
    } else {
        offlineMode = false;
    }

    lastFullRefresh = now();
    updateUI();
}

function updateUI() {
    renderCards();
    renderGatilhos();
    updateLeituraDetalhada();
    updateTicker();
}

// ====== MANUAL REFRESH ======
document.getElementById('refreshBtn').addEventListener('click', async () => {
    document.getElementById('refreshBtn').textContent = '↻ Atualizando...';
    await fetchAllIndicators();
    document.getElementById('refreshBtn').textContent = '↻ Atualizar';
    showToast('✔️ Dados atualizados', 2000);
});

(async function init() {
